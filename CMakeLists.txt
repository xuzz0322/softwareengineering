cmake_minimum_required(VERSION 3.20)
project(course_registration VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_search_module(MYSQL REQUIRED IMPORTED_TARGET mysqlclient mariadb libmariadb)
add_executable(registration backend/src/main.cpp backend/src/db.cpp backend/src/service.cpp)
target_include_directories(registration PRIVATE backend/include backend/vendor)
target_link_libraries(registration PRIVATE PkgConfig::MYSQL OpenSSL::Crypto OpenSSL::SSL Threads::Threads)
target_compile_definitions(registration PRIVATE CPPHTTPLIB_OPENSSL_SUPPORT)
target_compile_options(registration PRIVATE -Wall -Wextra -Wpedantic)
add_executable(domain_tests tests/domain_tests.cpp)
target_include_directories(domain_tests PRIVATE backend/include backend/vendor)
enable_testing()
add_test(NAME domain COMMAND domain_tests)
